🧪 TESTING FEATURES

Try These in Playground:

1. Simple Query
   query {
     dashboardSummary {
       users
       customers
     }
   }

2. With Variables
   query GetResource($r: String!) {
     resourceReport(resource: $r) {
       total
     }
   }
   Variables: {"r": "customers"}

3. Schema Introspection
   query {
     __schema {
       types {
         name
         description
       }
     }
   }

4. Type Details
   query {
     __type(name: "DashboardSummary") {
       fields {
         name
         description
       }
     }
   }
